/** * ============================================================================== * QS PRO v2.7 - Module AI Smart OCR & Tự Động Bóc Tách Cốt Thép TCVN * Cập nhật: Đầy đủ các hàm tính toán, chuẩn hóa định mức, kiểm tra chiều dài * thanh thép chuẩn 11.7m và phân loại nhóm cấu kiện (Móng, Cột, Dầm, Sàn, Lanh tô). * ============================================================================== */ class QSProSteelExtractorV27 { constructor() { // Bảng trọng lượng lý thuyết thép theo TCVN (kg/m) this.steelDensityTable = { 6: 0.222, 8: 0.395, 10: 0.617, 12: 0.888, 14: 1.208, 16: 1.578, 18: 1.998, 20: 2.466, 22: 2.984, 25: 3.855, 28: 4.834, 32: 6.313 }; this.defaultUnitPrices = { 'AF.61111': 18500000, // Thép móng 'AF.61113': 18500000, // Thép cột 'AF.61118': 18500000, // Thép dầm 'AF.61121': 18500000, // Thép sàn 'AF.61110': 18500000 // Thép chung/lanh tô }; } /** * Xử lý dữ liệu thô từ OCR toàn bộ 77 trang bản vẽ * @param {Array} ocrRawData - Mảng dữ liệu OCR đầu vào * @returns {Array} Mảng khối lượng chi tiết sau khi chuẩn hóa */ processOcrResult(ocrRawData) { if (!Array.isArray(ocrRawData) || ocrRawData.length === 0) { console.warn("QS PRO v2.7: Không có dữ liệu đầu vào từ OCR."); return []; } let parsedItems = ocrRawData.map((row, index) => { let diameter = this.extractDiameter(row.text || row.diameter); let length = parseFloat(row.length) || 11.7; let quantity = parseInt(row.quantity) || 1; let category = row.category || 'Dầm'; let elementName = row.elementName || 'Đang cập nhật'; // Kiểm tra cảnh báo nếu chiều dài vượt quá thanh tiêu chuẩn 11.7m let warningNote = length > 11.7 ? " (Cảnh báo: Lớn hơn 11.7m, cần kiểm tra nối chồng)" : ""; // Tính trọng lượng chuẩn xác theo TCVN (Kg đổi sang Tấn) let unitWeightPerMeter = this.steelDensityTable[diameter] || (Math.pow(diameter, 2) / 162); let totalWeightTon = (quantity * length * unitWeightPerMeter) / 1000; let normCode = this.getNormCode(category); let unitPrice = this.defaultUnitPrices[normCode] || 18500000; return { stt: index + 1, mucDinhMuc: normCode, tenCongViec: `Thép ${category} (${elementName}) d${diameter}mm - v2.7${warningNote}`, dvt: 'Tấn', soLuong: quantity, chieuDai: length, khoiLuong: parseFloat(totalWeightTon.toFixed(3)), donGia: unitPrice, thanhTien: parseFloat((totalWeightTon * unitPrice).toFixed(0)) }; }); return parsedItems; } /** * Trích xuất đường kính thép từ chuỗi văn bản (VD: d18, phi 12, D10) */ extractDiameter(text) { if (!text) return 18; let match = text.toString().match(/(?:d|phi)\s*(\d+)/i); return match ? parseInt(match[1]) : 18; } /** * Phân loại mã định mức theo hạng mục cấu kiện công trình */ getNormCode(category) { let catLower = (category || '').toLowerCase(); if (catLower.includes('móng') || catLower.includes('đài') || catLower.includes('cọc')) return 'AF.61111'; if (catLower.includes('cột')) return 'AF.61113'; if (catLower.includes('dầm')) return 'AF.61118'; if (catLower.includes('sàn')) return 'AF.61121'; return 'AF.61110'; // Lanh tô, cầu thang, cấu kiện phụ } /** * Xuất toàn bộ bảng thành định dạng mảng dữ liệu đẩy vào giao diện Web */ exportToTable(parsedData) { let totalWeight = parsedData.reduce((sum, item) => sum + item.khoiLuong, 0); let totalAmount = parsedData.reduce((sum, item) => sum + item.thanhTien, 0); return { items: parsedData, summary: { tongKhoiLuongTon: parseFloat(totalWeight.toFixed(3)), tongThanhTienVND: totalAmount } }; } } // ============================================================================== // KHỞI CHẠY VÀ KIỂM THỬ MODULE V2.7 // ============================================================================== const qsEngineV27 = new QSProSteelExtractorV27(); // Dữ liệu mẫu kiểm thử từ bản vẽ 77 trang (VD: Dầm D1) const sampleOcrInput = [ { category: 'Dầm', elementName: 'D1', text: 'Thép chủ d18', quantity: 12, length: 11.7 }, { category: 'Dầm', elementName: 'D1', text: 'Thép đai d8', quantity: 45, length: 1.2 } ]; let resultV27 = qsEngineV27.processOcrResult(sampleOcrInput); let finalTable = qsEngineV27.exportToTable(resultV27); console.log("=== KẾT QUẢ XUẤT BẢNG QS PRO v2.7 ==="); console.table(finalTable.items); console.log("Tổng khối lượng (Tấn):", finalTable.summary.tongKhoiLuongTon); console.log("Tổng thành tiền (VNĐ):", finalTable.summary.tongThanhTienVND.toLocaleString('vi-VN'));
